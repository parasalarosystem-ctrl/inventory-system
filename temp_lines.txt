    pattern = r'^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*[@$!%*?&])[A-Za-z\d@$!%*?&]{8,}$'
    return re.match(pattern, password)

    balance = item.balance or 0
    reorder = item.reorder_point or 5
    if balance <= 0:
        flash(f'Low Stock: "{item.item_description}" is Out of Stock.', 'warning')
    elif balance <= reorder:
        flash(f'Low Stock: "{item.item_description}" has only {balance} left.', 'warning')
